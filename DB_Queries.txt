create or replace PROCEDURE SR_MONITOR_DEL_90DAYS
as
BEGIN
DELETE from SR_MONITOR where REPLICATION_TIMESTAMP < sysdate -90;
commit;
END SR_MONITOR_DEL_90DAYS;
/

begin
sys.DBMS_SCHEDULER.create_job (
job_name => 'SR_MONITOR_DEL_90DAYS_JOB',
job_type => 'STORED_PROCEDURE',
job_action => 'SR_MONITOR_DEL_90DAYS',
start_date => SYSTIMESTAMP,
repeat_interval => 'FREQ=WEEKLY;BYDAY=SAT;BYHOUR=22',
end_date => NULL,
enabled => TRUE,
comments => 'SR MONITOR DELETE 90DAYS');
end;
/
